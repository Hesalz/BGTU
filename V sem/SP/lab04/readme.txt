regsvr32 OS13_HTCOM.dll
Обратный процесс
regsvr32 /u OS13_HTCOM.dll

".\x64\Debug\OS_0403.exe" "./HT.ht"
".\x64\Debug\OS_0404.exe" "./HT.ht"
